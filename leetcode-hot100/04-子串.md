## 1. 和为 K 的子数组（560）

**难度**：中等

**题目描述**：给你一个整数数组 `nums` 和一个整数 `k`，请你统计并返回该数组中和为 `k` 的子数组的个数。子数组是数组中元素的连续非空序列。

**原题链接**：https://leetcode.cn/problems/subarray-sum-equals-k/

**解题思路**：
- 前缀和 + 哈希表：`pre[i] - pre[j] = k` 表示区间 `(j, i]` 和为 k，即需要统计前面出现过的前缀和为 `pre[i] - k` 的次数
- 遍历时维护哈希表：key 为前缀和，value 为该前缀和出现次数；注意先累加答案再更新哈希表，并预存 `{0: 1}` 表示空前缀
- 时间复杂度 O(n)，空间复杂度 O(n)

**题解代码（Python）**：
```python
class Solution:
    def subarraySum(self, nums: List[int], k: int) -> int:
        count = {0: 1}
        pre, ans = 0, 0
        for num in nums:
            pre += num
            ans += count.get(pre - k, 0)
            count[pre] = count.get(pre, 0) + 1
        return ans
```

**题解代码（Java）**：
```java
class Solution {
    public int subarraySum(int[] nums, int k) {
        Map<Integer, Integer> count = new HashMap<>();
        count.put(0, 1);
        int pre = 0, ans = 0;
        for (int num : nums) {
            pre += num;
            ans += count.getOrDefault(pre - k, 0);
            count.merge(pre, 1, Integer::sum);
        }
        return ans;
    }
}
```

**题解代码（Go）**：
```go
func subarraySum(nums []int, k int) int {
    count := map[int]int{0: 1}
    pre, ans := 0, 0
    for _, num := range nums {
        pre += num
        ans += count[pre-k]
        count[pre]++
    }
    return ans
}
```

## 2. 滑动窗口最大值（239）

**难度**：困难

**题目描述**：给你一个整数数组 `nums`，有一个大小为 `k` 的滑动窗口从数组的最左侧移动到数组的最右侧。你只可以看到在滑动窗口内的 `k` 个数字，窗口每次向右移动一位。返回 滑动窗口中的最大值。

**原题链接**：https://leetcode.cn/problems/sliding-window-maximum/

**解题思路**：
- 单调递减双端队列（deque 存下标）：队头始终是当前窗口最大值的下标
- 入队前把队尾所有小于当前元素的值弹出（它们不可能再成为最大值）
- 队头下标超出窗口左边界时弹出队头；窗口形成后每次取队头元素
- 时间复杂度 O(n)（每个元素最多入队出队各一次），空间复杂度 O(k)

**题解代码（Python）**：
```python
class Solution:
    def maxSlidingWindow(self, nums: List[int], k: int) -> List[int]:
        deque, ans = collections.deque(), []
        for i, num in enumerate(nums):
            while deque and nums[deque[-1]] <= num:
                deque.pop()
            deque.append(i)
            if deque[0] <= i - k:
                deque.popleft()
            if i >= k - 1:
                ans.append(nums[deque[0]])
        return ans
```

**题解代码（Java）**：
```java
class Solution {
    public int[] maxSlidingWindow(int[] nums, int k) {
        int n = nums.length;
        Deque<Integer> deque = new ArrayDeque<>();
        int[] ans = new int[n - k + 1];
        for (int i = 0; i < n; i++) {
            while (!deque.isEmpty() && nums[deque.peekLast()] <= nums[i]) {
                deque.pollLast();
            }
            deque.offerLast(i);
            if (deque.peekFirst() <= i - k) {
                deque.pollFirst();
            }
            if (i >= k - 1) {
                ans[i - k + 1] = nums[deque.peekFirst()];
            }
        }
        return ans;
    }
}
```

**题解代码（Go）**：
```go
func maxSlidingWindow(nums []int, k int) []int {
    deque := []int{} // 存下标，对应值单调递减
    ans := []int{}

    for i,num:=range nums{
        for len(deque)>0&& num>=nums[deque[len(deque)-1]]{
            deque=deque[:len(deque)-1]
        }

        deque=append(deque,i)

        left:=i-k+1
        if deque[0]<left{
            deque=deque[1:]
        }

        if left>=0{
            ans=append(ans,nums[deque[0]])
        }
    }

    return ans
}
```

## 3. 最小覆盖子串（76）🔴🔴🔴🔴🔴🔴🔴

**难度**：困难

**题目描述**：给你一个字符串 `s`、一个字符串 `t`。返回 `s` 中涵盖 `t` 所有字符（包括重复字符）的最小子串。如果 `s` 中不存在涵盖 `t` 所有字符的子串，则返回空字符串 `""`。

**原题链接**：https://leetcode.cn/problems/minimum-window-substring/

**解题思路**：
- 滑动窗口：need 记录 t 中每个字符的需求数，window 记录窗口内计数，valid 记录满足需求的字符种数
- right 右移扩张窗口，当 valid == len(need)（所有字符都满足）时，尝试左移 left 收缩窗口并更新最小答案
- 收缩到不满足为止，继续扩张，直到 right 到末尾
- 时间复杂度 O(|s| + |t|)，空间复杂度 O(|Σ|)

**题解代码（Python）**：
```python
class Solution:
    def minWindow(self, s: str, t: str) -> str:
        need = collections.Counter(t)
        window = collections.defaultdict(int)
        valid = 0
        left = 0
        start, length = 0, float('inf')
        for right, ch in enumerate(s):
            if ch in need:
                window[ch] += 1
                if window[ch] == need[ch]:
                    valid += 1
            while valid == len(need):
                if right - left + 1 < length:
                    start, length = left, right - left + 1
                if s[left] in need:
                    if window[s[left]] == need[s[left]]:
                        valid -= 1
                    window[s[left]] -= 1
                left += 1
        return "" if length == float('inf') else s[start:start + length]
```

**题解代码（Java）**：
```java
class Solution {
    public String minWindow(String s, String t) {
        Map<Character, Integer> need = new HashMap<>();
        Map<Character, Integer> window = new HashMap<>();
        for (char c : t.toCharArray()) need.merge(c, 1, Integer::sum);
        int valid = 0, left = 0;
        int start = 0, length = Integer.MAX_VALUE;
        for (int right = 0; right < s.length(); right++) {
            char ch = s.charAt(right);
            if (need.containsKey(ch)) {
                window.merge(ch, 1, Integer::sum);
                if (window.get(ch).intValue() == need.get(ch).intValue()) valid++;
            }
            while (valid == need.size()) {
                if (right - left + 1 < length) {
                    start = left;
                    length = right - left + 1;
                }
                char lc = s.charAt(left);
                left++;
                if (need.containsKey(lc)) {
                    if (window.get(lc).intValue() == need.get(lc).intValue()) valid--;
                    window.merge(lc, -1, Integer::sum);
                }
            }
        }
        return length == Integer.MAX_VALUE ? "" : s.substring(start, start + length);
    }
}
```

**题解代码（Go）**：
```go
func minWindow(s string, t string) string {
    need := map[byte]int{}
    for i := 0; i < len(t); i++ {
        need[t[i]]++
    }
    window := map[byte]int{}
    valid, left := 0, 0
    start, length := 0, math.MaxInt32
    for right := 0; right < len(s); right++ {
        ch := s[right]
        if _, ok := need[ch]; ok {
            window[ch]++
            if window[ch] == need[ch] {
                valid++
            }
        }
        for valid == len(need) {
            if right-left+1 < length {
                start, length = left, right-left+1
            }
            lc := s[left]
            left++
            if _, ok := need[lc]; ok {
                if window[lc] == need[lc] {
                    valid--
                }
                window[lc]--
            }
        }
    }
    if length == math.MaxInt32 {
        return ""
    }
    return s[start : start+length]
}
```
