## 1. 数组中的第K个最大元素（215）

**难度**：中等

**题目描述**：给定整数数组 `nums` 和整数 `k`，请返回数组中第 `k` 个最大的元素。请注意，你需要找的是数组排序后的第 `k` 个最大的元素，而不是第 `k` 个不同的元素。你必须设计并实现时间复杂度为 `O(n)` 的算法解决此问题。

**原题链接**：https://leetcode.cn/problems/kth-largest-element-in-an-array/

**解题思路**：
- 方法一（快速选择）：借鉴快排的 partition，随机选 pivot 分区后判断第 k 大落在哪一侧，只递归一侧，平均时间 O(n)
  - 第 k 大 = 排序后下标 `n - k`
- 方法二（小顶堆）：维护大小为 k 的小顶堆，遍历数组，堆满且当前数大于堆顶则替换；最终堆顶即答案，时间 O(n log k)
- 时间复杂度平均 O(n)，空间 O(1)（原地）或 O(k)

**题解代码（Python）**：
```python
import random

class Solution:
    def findKthLargest(self, nums: List[int], k: int) -> int:
        target = len(nums) - k  # 排序后该元素的下标
        left, right = 0, len(nums) - 1
        while True:
            idx = self.partition(nums, left, right)
            if idx == target:
                return nums[idx]
            elif idx < target:
                left = idx + 1
            else:
                right = idx - 1

    def partition(self, nums, left, right):
        # 随机选取 pivot，避免最坏情况
        rand = random.randint(left, right)
        nums[rand], nums[right] = nums[right], nums[rand]
        pivot = nums[right]
        i = left
        for j in range(left, right):
            if nums[j] < pivot:
                nums[i], nums[j] = nums[j], nums[i]
                i += 1
        nums[i], nums[right] = nums[right], nums[i]
        return i
```

**题解代码（Java）**：
```java
import java.util.Random;

class Solution {
    private final Random random = new Random();

    public int findKthLargest(int[] nums, int k) {
        int target = nums.length - k;
        int left = 0, right = nums.length - 1;
        while (true) {
            int idx = partition(nums, left, right);
            if (idx == target) return nums[idx];
            else if (idx < target) left = idx + 1;
            else right = idx - 1;
        }
    }

    private int partition(int[] nums, int left, int right) {
        int rand = left + random.nextInt(right - left + 1);
        swap(nums, rand, right);
        int pivot = nums[right], i = left;
        for (int j = left; j < right; j++) {
            if (nums[j] < pivot) swap(nums, i++, j);
        }
        swap(nums, i, right);
        return i;
    }

    private void swap(int[] nums, int a, int b) {
        int tmp = nums[a];
        nums[a] = nums[b];
        nums[b] = tmp;
    }
}
```

**题解代码（Go）**：

```go
            right = idx - 1
        }
    }

func partition(nums []int, left, right int) int {
    // 随机选取 pivot，避免最坏情况
    rand := left + rand.Intn(right-left+1)
    nums[rand], nums[right] = nums[right], nums[rand]
    pivot, i := nums[right], left
    for j := left; j < right; j++ {
        if nums[j] < pivot {
            nums[i], nums[j] = nums[j], nums[i]
            i++
        }
    }
    nums[i], nums[right] = nums[right], nums[i]
    return i
}
```

## 2. 前 K 个高频元素（347）

**难度**：中等

**题目描述**：给你一个整数数组 `nums` 和一个整数 `k`，请你返回其中出现频率前 `k` 高的元素。你可以按 任意顺序 返回答案。请你设计时间复杂度优于 `O(n log n)` 的算法解决此问题。

**原题链接**：https://leetcode.cn/problems/top-k-frequent-elements/

**解题思路**：
- 哈希表统计频率，维护大小为 k 的小顶堆（按频率），频率大于堆顶则替换堆顶；最终堆内即前 k 高频元素，时间 O(n log k)
- 进阶（桶排序达 O(n)）：按频率分桶（桶下标即频率），从高频桶往低收集 k 个
- 时间复杂度 O(n)（桶排序版），空间 O(n)

**题解代码（Python）**：
```python
class Solution:
    def topKFrequent(self, nums: List[int], k: int) -> List[int]:
        count = Counter(nums)
        buckets = [[] for _ in range(len(nums) + 1)]  # 桶排序
        for num, freq in count.items():
            buckets[freq].append(num)
        ans = []
        for freq in range(len(buckets) - 1, 0, -1):
            for num in buckets[freq]:
                ans.append(num)
                if len(ans) == k:
                    return ans
        return ans
```

**题解代码（Java）**：
```java
class Solution {
    public int[] topKFrequent(int[] nums, int k) {
        Map<Integer, Integer> count = new HashMap<>();
        for (int num : nums) count.merge(num, 1, Integer::sum);
        List<List<Integer>> buckets = new ArrayList<>();
        for (int i = 0; i <= nums.length; i++) buckets.add(new ArrayList<>());
        for (Map.Entry<Integer, Integer> e : count.entrySet()) {
            buckets.get(e.getValue()).add(e.getKey());
        }
        int[] ans = new int[k];
        int idx = 0;
        for (int freq = buckets.size() - 1; freq > 0 && idx < k; freq--) {
            for (int num : buckets.get(freq)) {
                ans[idx++] = num;
                if (idx == k) break;
            }
        }
        return ans;
    }
}
```

**题解代码（Go）**：
```go
func topKFrequent(nums []int, k int) []int {
    count := map[int]int{}
    for _, num := range nums {
        count[num]++
    }
    buckets := make([][]int, len(nums)+1) // 桶排序

        buckets[freq] = append(buckets[freq], num)
    }
    ans := []int{}
    for freq := len(buckets) - 1; freq > 0; freq-- {
        for _, num := range buckets[freq] {
            ans = append(ans, num)
            if len(ans) == k {
                return ans
            }
        }
    }
    return ans
}
```

## 3. 数据流的中位数（295）

**难度**：困难

**题目描述**：中位数是有序整数列表中的中间值。如果列表的长度为偶数，则中位数是两个中间值的平均值。例如 `arr = [2,3,4]` 的中位数是 `3`；`arr = [2,3]` 的中位数是 `(2 + 3) / 2 = 2.5`。实现 `MedianFinder` 类:`MedianFinder()` 初始化 MedianFinder 对象；`void addNum(int num)` 将整数 num 添加到数据结构中；`double findMedian()` 返回到目前为止所有元素的中位数。

**原题链接**：https://leetcode.cn/problems/find-median-from-data-stream/

**解题思路**：
- 对顶堆：大顶堆 `lo` 存较小的一半（堆顶为该半最大值），小顶堆 `hi` 存较大的一半（堆顶为该半最小值）
- 保持 `len(lo) == len(hi)` 或 `len(lo) == len(hi) + 1`
- addNum 流程：先压入 lo，弹出 lo 堆顶压入 hi；若 hi 比 lo 大则把 hi 堆顶移回 lo，两步保证平衡有序
- findMedian：奇数取 lo 堆顶；偶数取两堆顶平均
- addNum O(log n)，findMedian O(1)

**题解代码（Python）**：
```python
class MedianFinder:
    def __init__(self):
        self.lo = []  # 大顶堆（存负数）
        self.hi = []  # 小顶堆

    def addNum(self, num: int) -> None:
        heapq.heappush(self.lo, -num)
        heapq.heappush(self.hi, -heapq.heappop(self.lo))
        if len(self.hi) > len(self.lo):
            heapq.heappush(self.lo, -heapq.heappop(self.hi))

    def findMedian(self) -> float:
        if len(self.lo) > len(self.hi):
            return -self.lo[0]
        return (-self.lo[0] + self.hi[0]) / 2
```

**题解代码（Java）**：
```java
class MedianFinder {
    private final PriorityQueue<Integer> lo = new PriorityQueue<>((a, b) -> b - a);
    private final PriorityQueue<Integer> hi = new PriorityQueue<>();

    public MedianFinder() {}

    public void addNum(int num) {
        lo.offer(num);
        hi.offer(lo.poll());
        if (hi.size() > lo.size()) {
            lo.offer(hi.poll());
        }
    }

    public double findMedian() {
        if (lo.size() > hi.size()) return lo.peek();
        return (lo.peek() + hi.peek()) / 2.0;
    }
}
```

**题解代码（Go）**：
```go
type MedianFinder struct {
    lo, hi hp // lo 大顶堆（存负数），hi 小顶堆
}

type hp []int

func (h hp) Len() int            { return len(h) }
func (h hp) Less(i, j int) bool  { return h[i] < h[j] }
func (h hp) Swap(i, j int)       { h[i], h[j] = h[j], h[i] }
func (h *hp) Push(x interface{}) { *h = append(*h, x.(int)) }
func (h *hp) Pop() interface{} {
    old := *h
    n := len(old)
    x := old[n-1]
    *h = old[:n-1]
    return x
}

func Constructor() MedianFinder {
    return MedianFinder{}
}

func (mf *MedianFinder) AddNum(num int) {
    // 大顶堆用负数模拟
    heap.Push(&mf.lo, -num)
    maxOfLo, _ := heap.Pop(&mf.lo).(-int)
    heap.Push(&mf.hi, maxOfLo)
    if mf.hi.Len() > mf.lo.Len() {
        minOfHi, _ := heap.Pop(&mf.hi).(int)
        heap.Push(&mf.lo, -minOfHi)
    }
}

func (mf *MedianFinder) FindMedian() float64 {
    if mf.lo.Len() > mf.hi.Len() {
        return float64(-mf.lo[0])
    }
    return float64(-mf.lo[0]+mf.hi[0]) / 2
}
```
