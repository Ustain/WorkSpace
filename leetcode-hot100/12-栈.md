## 1. 有效的括号（20）

**难度**：简单

**题目描述**：给定一个只包括 `'('`，`')'`，`'{'`，`'}'`，`'['`，`']'` 的字符串 `s`，判断字符串是否有效。有效字符串需满足：左括号必须用相同类型的右括号闭合；左括号必须以正确的顺序闭合；每个右括号都有一个对应的相同类型的左括号。

**原题链接**：https://leetcode.cn/problems/valid-parentheses/

**解题思路**：
- 栈匹配：左括号入栈；右括号弹出栈顶，必须是配对的左括号
- 用哈希表存 `右括号 → 左括号`；字符串为有效时栈必须为空（长度为奇数直接 false）
- 时间复杂度 O(n)，空间 O(n)

**题解代码（Python）**：
```python
class Solution:
    def isValid(self, s: str) -> bool:
        if len(s) % 2 == 1:
            return False
        pairs = {')': '(', ']': '[', '}': '{'}
        stack = []
        for ch in s:
            if ch in pairs:
                if not stack or stack[-1] != pairs[ch]:
                    return False
                stack.pop()
            else:
                stack.append(ch)
        return not stack
```

**题解代码（Java）**：
```java
class Solution {
    public boolean isValid(String s) {
        if (s.length() % 2 == 1) return false;
        Map<Character, Character> pairs = Map.of(')', '(', ']', '[', '}', '{');
        Deque<Character> stack = new ArrayDeque<>();
        for (char ch : s.toCharArray()) {
            if (pairs.containsKey(ch)) {
                if (stack.isEmpty() || stack.peek() != pairs.get(ch)) return false;
                stack.pop();
            } else {
                stack.push(ch);
            }
        }
        return stack.isEmpty();
    }
}
```

**题解代码（Go）**：
```go
func isValid(s string) bool {
    if len(s)%2 == 1 {
        return false
    }
    pairs := map[byte]byte{')': '(', ']': '[', '}': '{'}
    stack := []byte{}

        ch := s[i]
        if left, ok := pairs[ch]; ok {
            if len(stack) == 0 || stack[len(stack)-1] != left {
                return false
            }
            stack = stack[:len(stack)-1]
        } else {
            stack = append(stack, ch)
        }
    }
    return len(stack) == 0
}
```

## 2. 最小栈（155）

**难度**：中等

**题目描述**：设计一个支持 `push` ，`pop` ，`top` 操作，并能在常数时间内检索到最小元素的栈。实现 `MinStack` 类:`MinStack()` 初始化堆栈对象；`void push(int val)` 将元素val推入堆栈；`void pop()` 删除堆栈顶部的元素；`int top()` 获取堆栈顶部的元素；`int getMin()` 获取堆栈中的最小元素。

**原题链接**：https://leetcode.cn/problems/min-stack/

**解题思路**：
- 辅助栈与主栈同步进出：辅助栈顶始终保存"当前栈内所有元素的最小值"
- push 时，辅助栈压入 `min(新元素, 辅助栈顶)`；pop 时两栈同步弹出
- 每个操作 O(1)，空间 O(n)

**题解代码（Python）**：
```python
class MinStack:
    def __init__(self):
        self.stack = []
        self.min_stack = [math.inf]

    def push(self, val: int) -> None:
        self.stack.append(val)
        self.min_stack.append(min(val, self.min_stack[-1]))

    def pop(self) -> None:
        self.stack.pop()
        self.min_stack.pop()

    def top(self) -> int:
        return self.stack[-1]

    def getMin(self) -> int:
        return self.min_stack[-1]
```

**题解代码（Java）**：
```java
class MinStack {
    private Deque<Integer> stack = new ArrayDeque<>();
    private Deque<Integer> minStack = new ArrayDeque<>();

    public MinStack() {
        minStack.push(Integer.MAX_VALUE);
    }

    public void push(int val) {
        stack.push(val);
        minStack.push(Math.min(val, minStack.peek()));
    }

    public void pop() {
        stack.pop();
        minStack.pop();
    }

    public int top() {
        return stack.peek();
    }

    public int getMin() {
        return minStack.peek();
    }
}
```

**题解代码（Go）**：
```go
type MinStack struct {
    stack    []int
    minStack []int
}

func Constructor() MinStack {
    return MinStack{minStack: []int{math.MaxInt64}}
}

func (s *MinStack) Push(val int) {
    s.stack = append(s.stack, val)
    top := s.minStack[len(s.minStack)-1]
    if val < top {
        top = val
    }
    s.minStack = append(s.minStack, top)
}

func (s *MinStack) Pop() {
    s.stack = s.stack[:len(s.stack)-1]
    s.minStack = s.minStack[:len(s.minStack)-1]
}

func (s *MinStack) Top() int {
    return s.stack[len(s.stack)-1]
}

func (s *MinStack) GetMin() int {
    return s.minStack[len(s.minStack)-1]
}
```

## 3. 字符串解码（394）

**难度**：中等

**题目描述**：给定一个经过编码的字符串，返回它解码后的字符串。编码规则为: `k[encoded_string]`，表示其中方括号内部的 encoded_string 正好重复 k 次。保证输入符合编码规则，保证输入的方括号总是有效，保证 k 为正数。

**原题链接**：https://leetcode.cn/problems/decode-string/

**解题思路**：
- 双栈：数字栈存重复次数、字符串栈存括号前的已构建结果
- 遇到数字累积解析多位数；遇到 `[` 把当前数字和当前字符串入栈并重置；遇到 `]` 弹出重复次数 prevNum 与之前字符串 prevStr，`当前串 = prevStr + 当前串 × prevNum`
- 时间复杂度 O(n · maxK·|s|)（输出规模），空间同输出

**题解代码（Python）**：
```python
class Solution:
    def decodeString(self, s: str) -> str:
        num_stack, str_stack = [], []
        cur, num = [], 0
        for ch in s:
            if ch.isdigit():
                num = num * 10 + int(ch)
            elif ch == '[':
                num_stack.append(num)
                str_stack.append(cur)
                num = 0
                cur = []
            elif ch == ']':
                repeat = num_stack.pop()
                prev = str_stack.pop()
                cur = prev + cur * repeat
            else:
                cur.append(ch)
        return ''.join(cur)
```

**题解代码（Java）**：
```java
class Solution {
    public String decodeString(String s) {
        Deque<Integer> numStack = new ArrayDeque<>();
        Deque<StringBuilder> strStack = new ArrayDeque<>();
        StringBuilder cur = new StringBuilder();
        int num = 0;
        for (char ch : s.toCharArray()) {
            if (Character.isDigit(ch)) {
                num = num * 10 + (ch - '0');
            } else if (ch == '[') {
                numStack.push(num);
                strStack.push(cur);
                num = 0;
                cur = new StringBuilder();
            } else if (ch == ']') {
                int repeat = numStack.pop();
                StringBuilder prev = strStack.pop();
                for (int i = 0; i < repeat; i++) prev.append(cur);
                cur = prev;
            } else {
                cur.append(ch);
            }
        }
        return cur.toString();
    }
}
```

**题解代码（Go）**：
```go
func decodeString(s string) string {
    numStack, strStack := []int{}, []string{}
    cur := ""
    num := 0
    for i := 0; i < len(s); i++ {
        ch := s[i]
        if ch >= '0' && ch <= '9' {
            num = num*10 + int(ch-'0')
        } else if ch == '[' {
            numStack = append(numStack, num)
            strStack = append(strStack, cur)
            num = 0
            cur = ""
        } else if ch == ']' {
            repeat := numStack[len(numStack)-1]
            numStack = numStack[:len(numStack)-1]
            prev := strStack[len(strStack)-1]
            strStack = strStack[:len(strStack)-1]
            cur = prev + strings.Repeat(cur, repeat)
        } else {
            cur += string(ch)
        }
    }
    return cur
}
```

## 4. 每日温度（739）

**难度**：中等

**题目描述**：给定一个整数数组 `temperatures`，表示每天的温度，返回一个数组 `answer`，其中 `answer[i]` 是指对于第 `i` 天，下一个更高温度出现在几天后。如果气温在这之后都不会升高，请在该位置用 `0` 来代替。

**原题链接**：https://leetcode.cn/problems/daily-temperatures/

**解题思路**：
- 单调栈（存下标，栈内温度递减）：当前温度高于栈顶温度时，栈顶元素的"下一个更高温度"就是当前天
- 不断弹出栈顶并计算下标差；处理完当前元素入栈
- 每个元素最多进出栈一次，时间复杂度 O(n)，空间 O(n)

**题解代码（Python）**：
```python
class Solution:
    def dailyTemperatures(self, temperatures: List[int]) -> List[int]:
        n = len(temperatures)
        answer = [0] * n
        stack = []  # 存下标，对应温度单调递减
        for i in range(n):
            while stack and temperatures[i] > temperatures[stack[-1]]:
                j = stack.pop()
                answer[j] = i - j
            stack.append(i)
        return answer
```

**题解代码（Java）**：
```java
class Solution {
    public int[] dailyTemperatures(int[] temperatures) {
        int n = temperatures.length;
        int[] answer = new int[n];
        Deque<Integer> stack = new ArrayDeque<>();
        for (int i = 0; i < n; i++) {
            while (!stack.isEmpty() && temperatures[i] > temperatures[stack.peek()]) {
                int j = stack.pop();
                answer[j] = i - j;
            }
            stack.push(i);
        }
        return answer;
    }
}
```


## 5. 柱状图中最大的矩形（84）

**难度**：困难

**题目描述**：给定 n 个非负整数，用来表示柱状图中各个柱子的高度。每个柱子彼此相邻，且宽度为 1。求在该柱状图中，能够勾勒出来的矩形的最大面积。

**原题链接**：https://leetcode.cn/problems/largest-rectangle-in-histogram/

**解题思路**：
- 单调递增栈：对每根柱子，找到左右两边第一个比它矮的柱子，宽度 × 高度即为以它为高的最大矩形
- 遍历中当当前柱子矮于栈顶时，栈顶柱子出栈——此时其右边界是当前柱、左边界是新的栈顶（均在两侧最近矮处）
- 哨兵技巧：数组首尾加高度 0（或末尾补 0），保证所有柱子最终出栈
- 时间复杂度 O(n)，空间 O(n)

**题解代码（Python）**：
```python
class Solution:
    def largestRectangleArea(self, heights: List[int]) -> int:
        stack = [-1]  # 哨兵：栈底存 -1 下标
        max_area = 0
        heights.append(0)  # 尾部哨兵，触发全部出栈
        for i, h in enumerate(heights):
            while stack[-1] != -1 and heights[stack[-1]] >= h:
                height = heights[stack.pop()]
                width = i - stack[-1] - 1
                max_area = max(max_area, height * width)
            stack.append(i)
        return max_area
```

**题解代码（Java）**：
```java
class Solution {
    public int largestRectangleArea(int[] heights) {
        int n = heights.length;
        Deque<Integer> stack = new ArrayDeque<>();
        stack.push(-1);
        int maxArea = 0;
        for (int i = 0; i <= n; i++) {
            int h = (i == n) ? 0 : heights[i];  // 尾部哨兵
            while (stack.peek() != -1 && heights[stack.peek()] >= h) {
                int height = heights[stack.pop()];
                int width = i - stack.peek() - 1;
                maxArea = Math.max(maxArea, height * width);
            }
            stack.push(i);
        }
        return maxArea;
    }
}
```

**题解代码（Go）**：
```go
func largestRectangleArea(heights []int) int {
    stack := []int{-1} // 哨兵：栈底存 -1 下标
    maxArea := 0
    heights = append(heights, 0) // 尾部哨兵，触发全部出栈
    for i, h := range heights {
        for stack[len(stack)-1] != -1 && heights[stack[len(stack)-1]] >= h {
            height := heights[stack[len(stack)-1]]
            stack = stack[:len(stack)-1]
            width := i - stack[len(stack)-1] - 1
            if height*width > maxArea {
                maxArea = height * width
            }
        }
        stack = append(stack, i)
    }
    return maxArea
}
```
